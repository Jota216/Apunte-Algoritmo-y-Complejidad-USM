#1 Bungee Builder
Listado 1 #1
Problema: Se tienen n montañas alineadas con alturas h_i (1 <= n <= 10^6). Se instala un puente de bungee entre dos picos p y q (p < q). La caída libre es min(h_p, h_q) - min_{p < k < q}(h_k). Si entremedio hay montañas tan altas o más que los extremos, o no hay valle, el salto no es viable. Calcular la máxima altura de salto en la cordillera.
Técnica: Pila Monótona / Barrido por extremos. Se identifica el pico más alto absoluto y se procesa hacia la izquierda y derecha manteniendo una pila con alturas decrecientes y valles intermedios para maximizar la caída en O(N).
Tiempo: O(N) Espacio: O(N) 
#include <bits/stdc++.h>
using namespace std;

// Pila de pares: (altura_montaña, salto_acumulado_desde_el_valle)
int salto_max = 0;
vector<int> cerros;
stack<pair<int,int>> s;

void calcularSalto(int i) {
    int altura_salto_max = 0;
    // Si la montaña actual es más alta que la del tope, calculamos el salto posible
    while (!s.empty() && cerros[i] >= s.top().first) {
        altura_salto_max = max(cerros[i] - s.top().first + s.top().second, altura_salto_max);
        s.pop();
    }
    s.push({cerros[i], altura_salto_max});
    salto_max = max(salto_max, altura_salto_max);
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n;
    if (!(cin >> n)) return 0;
    cerros.resize(n);
    int max_alt = -1, max_pos = 0;
    for (int i = 0; i < n; i++) {
        cin >> cerros[i];
        if (cerros[i] > max_alt) {
            max_alt = cerros[i];
            max_pos = i;
        }
    }
    // Barrido hacia la izquierda desde el pico global
    s.push({max_alt, 0});
    for (int i = max_pos - 1; i >= 0; i--) calcularSalto(i);

    // Vaciar y barrido hacia la derecha desde el pico global
    while (!s.empty()) s.pop();
    s.push({max_alt, 0});
    for (int i = max_pos + 1; i < n; i++) calcularSalto(i);

    cout << salto_max << "\n";
    return 0;
}

Técnica: Tablas Hash / std::set. Se almacenan todas las palabras leídas en un vector y en un std::set. Se prueban todos los pares i != j formando w_i + w_j, guardándolas en un set ordenado para eliminar duplicados y ordenar automáticamente.
Tiempo: O(N^2 * L) Espacio: O(N * L) 
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    set<string> cws;
    vector<string> ws;
    string w;
    while (cin >> w) ws.push_back(w);

    int n = ws.size();
    // Probar todas las combinaciones de pares de palabras distintas
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            if (i == j) continue;
            cws.insert(ws[i] + ws[j]); // std::set garantiza unicidad y orden lexicográfico
        }
    }
    for (const string& s : cws) cout << s << "\n";
    return 0;
}
#3 Game of Throwns
Listado 1 #3
Problema: n niños (0 a n-1) en ronda se lanzan un huevo. Se dan k comandos: un entero d lanza el huevo d posiciones en la ronda (positivo horario, negativo antihorario); 'undo m' deshace los últimos m lanzamientos. Determinar la posición final del huevo.
Técnica: Simulación con Pila (std::vector). Cada lanzamiento exitoso empuja la posición absoluta a una pila. El comando 'undo m' hace m pops. El operador módulo `(pos % n + n) % n` maneja giros negativos.
Tiempo: O(K) Espacio: O(K) 
#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, k;
    if (!(cin >> n >> k)) return 0;
    vector<int> history;
    history.push_back(0); // Posición inicial: niño 0

    for (int i = 0; i < k; i++) {
        string s;
        cin >> s;
        if (s == "undo") {
            int m;
            cin >> m;
            while (m-- && history.size() > 1) history.pop_back(); // Deshacer m pasos
        } else {
            int throw_dist = stoi(s);
            int new_pos = (history.back() + throw_dist) % n;
            if (new_pos < 0) new_pos += n; // Corrección para residuos negativos
            history.push_back(new_pos);
        }
    }
    cout << history.back() << "\n";
    return 0;
}
#4 Integer Lists
Listado 1 #4
Problema: Interpretar un programa en lenguaje BAPC sobre una lista de números. 'R' invierte el orden de la lista, 'D' descarta el primer elemento. Si se aplica 'D' sobre una lista vacía, se debe reportar 'error'.
Técnica: Doble Cola (std::deque) con bandera de reversa. Invertir la lista se simula alternando un booleano `rev` en O(1). 'D' hace `pop_front` si `!rev`, o `pop_back` si `rev`. Al final se imprime en la dirección correspondiente.
Tiempo: O(|P| + N) Espacio: O(N) 
#include <bits/stdc++.h>
using namespace std;

void solve() {
    string p; cin >> p;
    int n; cin >> n;
    string arr_str; cin >> arr_str;

    deque<int> dq;
    stringstream ss(arr_str);
    char ch;
    int val;
    ss >> ch; // descartar '['
    for (int i = 0; i < n; i++) {
        ss >> val;
        dq.push_back(val);
        if (i < n - 1) ss >> ch; // descartar ','
    }

    bool rev = false, error = false;
    for (char c : p) {
        if (c == 'R') {
            rev = !rev; // O(1) en vez de invertir el arreglo
        } else if (c == 'D') {
            if (dq.empty()) { error = true; break; }
            if (rev) dq.pop_back();
            else dq.pop_front();
        }
    }

    if (error) {
        cout << "error\n";
    } else {
        cout << "[";
        while (!dq.empty()) {
            if (!rev) {
                cout << dq.front();
                dq.pop_front();
            } else {
                cout << dq.back();
                dq.pop_back();
            }
            if (!dq.empty()) cout << ",";
        }
        cout << "]\n";
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    if (cin >> t) while (t--) solve();
    return 0;
}

#6 Sim
Listado 1 #6
Problema: Simulación de un editor de texto ante teclas especiales: '[' mueve el cursor al inicio del texto (Home), ']' lo mueve al final (End), '<' elimina el carácter anterior al cursor (Backspace). Cadena de hasta 10^6 caracteres.
Técnica: Lista Doblemente Enlazada (`std::list<char>`) con iterador de cursor. Insertar caracteres antes del cursor y borrar con `erase` toma O(1) tiempo por operación, evitando reubicaciones en memoria de std::string.
Tiempo: O(|S|) Espacio: O(|S|) 
#include <iostream>
#include <string>
#include <list>
using namespace std;

void solve() {
    string line;
    if (!getline(cin, line)) return;
    list<char> text;
    auto it = text.end(); // El cursor comienza al final

    for (char c : line) {
        if (c == '[') {
            it = text.begin(); // Home: cursor al inicio
        } else if (c == ']') {
            it = text.end();   // End: cursor al final
        } else if (c == '<') {
            if (it != text.begin()) {
                auto to_erase = prev(it);
                text.erase(to_erase); // Backspace: borra carácter previo
            }
        } else {
            text.insert(it, c); // Inserta carácter en la posición del cursor en O(1)
        }
    }
    for (char c : text) cout << c;
    cout << "\n";
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    if (cin >> t) {
        string dummy;
        getline(cin, dummy);
        while (t--) solve();
    }
    return 0;
}
#7 Srednji
Listado 1 #7
Problema: Se da una permutación de n enteros de 1 a n y un valor B. Contar la cantidad de subarreglos contiguos de longitud impar que tengan al número B como su mediana.
Técnica: Prefix Sums Relativos + Hash Map / Vector de frecuencias. Reemplazar cada valor x: +1 si x > B, -1 si x < B, 0 si x == B. Se calculan las sumas acumuladas a la derecha de B y se cuenta cuántas veces cada suma ocurre. Luego se emparejan con sumas acumuladas a la izquierda para que sumen 0.
Tiempo: O(N) Espacio: O(N) 
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int n, b;
    if (!(cin >> n >> b)) return 0;
    vector<int> a(n);
    int b_pos = -1;
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        if (a[i] == b) b_pos = i;
    }

    // frequencies de balances hacia la derecha (offset n para evitar índices negativos)
    vector<int> freq(2 * n + 1, 0);
    int sum = 0;
    freq[n]++; // Balance 0 inicial en la posición de b
    for (int i = b_pos + 1; i < n; i++) {
        sum += (a[i] > b) ? 1 : -1;
        freq[sum + n]++;
    }

    long long count = 0;
    sum = 0;
    // Emparejar desde b hacia la izquierda
    for (int i = b_pos; i >= 0; i--) {
        if (i < b_pos) sum += (a[i] > b) ? 1 : -1;
        // Para que la mediana sea B, el balance total debe ser 0: sum_der == -sum_izq
        count += freq[-sum + n];
    }

    cout << count << "\n";
    return 0;
}
#8 Stock Prices
Listado 1 #8
Problema: Libro de órdenes de bolsa. Se introducen órdenes 'buy' (compra) y 'sell' (venta) con precio y volumen. Si buy_price >= sell_price, se ejecutan compras al sell_price. En cada evento imprimir: ask (menor precio de venta), bid (mayor precio de compra) y stock (último precio de transacción).
Técnica: Colas de Prioridad / std::map. Mantener las órdenes de compra en un max-heap (o map descendente) y las de venta en un min-heap (o map ascendente). Cruzar órdenes mientras el precio de compra sea mayor o igual al de venta.
Tiempo: O(N log N) Espacio: O(N) 
#include <iostream>
#include <map>
#include <string>
#include <algorithm>
using namespace std;

void solve() {
    int n; cin >> n;
    map<int, int> buy_orders;                 // Clave: precio (ascendente). rbegin() es el mayor
    map<int, int, greater<int>> sell_orders;  // Clave: precio (descendente). rbegin() es el menor
    int last_trade = -1;

    for (int i = 0; i < n; i++) {
        string type, dummy;
        int shares, price;
        cin >> type >> shares >> dummy >> dummy >> price;

        if (type == "buy") buy_orders[price] += shares;
        else sell_orders[price] += shares;

        // Ejecutar transacciones mientras la oferta de compra iguale o supere la de venta
        while (!buy_orders.empty() && !sell_orders.empty()) {
            auto highest_buy = prev(buy_orders.end());
            auto lowest_sell = prev(sell_orders.end());

            if (highest_buy->first >= lowest_sell->first) {
                last_trade = lowest_sell->first; // Se ejecuta al precio de venta
                int trade_shares = min(highest_buy->second, lowest_sell->second);
                highest_buy->second -= trade_shares;
                lowest_sell->second -= trade_shares;
                if (highest_buy->second == 0) buy_orders.erase(highest_buy);
                if (lowest_sell->second == 0) sell_orders.erase(lowest_sell);
            } else break;
        }

        // Salida: ask (menor venta), bid (mayor compra), stock (último trade)
        if (!sell_orders.empty()) cout << prev(sell_orders.end())->first << " ";
        else cout << "- ";
        if (!buy_orders.empty()) cout << prev(buy_orders.end())->first << " ";
        else cout << "- ";
        if (last_trade != -1) cout << last_trade << "\n";
        else cout << "-\n";
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    if (cin >> t) while (t--) solve();
    return 0;
}

#include <iostream>
#include <vector>
#include <string>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    vector<string> b(3);
    for (int i = 0; i < 3; i++) cin >> b[i];

    auto wins = [&](char c) {
        for (int i = 0; i < 3; i++) {
            if (b[i][0] == c && b[i][1] == c && b[i][2] == c) return true; // Filas

}

#11 Flujos Parametrizados en Grafos

Problema: Se tienen n nodos (1 <= n <= 10^6). Cada nodo tiene id y un valor crítico lambda (número real con hasta 6 decimales) donde transiciona. Agrupar los identificadores por valor de lambda y mostrarlos en orden ascendente de lambda: 'lambda: id1 id2 ...' con ids ordenados y formato de lambda eliminando ceros innecesarios (al menos 1 decimal).
Técnica: Tablas Hash / `std::map` ordenado con escala entera. Para evitar imprecisión de floats, escalar lambda multiplicando por 10^6 a `long long`. Formatear la cadena decimal podando ceros redundantes a la derecha pero conservando al menos un decimal si es entero.
Tiempo: O(n log n) Espacio: O(n) 
#include <iostream>
#include <vector>
#include <string>
#include <map>
#include <algorithm>
#include <cmath>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    if (!(cin >> n)) return 0;

    // Escalar lambda por 10^6 para usar enteros exactos como clave de ordenamiento
    map<long long, vector<int>> groups;
    for (int i = 0; i < n; i++) {
        int id;
        double lambda;
        cin >> id >> lambda;
        long long key = llround(lambda * 1e6);
        groups[key].push_back(id);
    }

    for (auto& [key, ids] : groups) {
        sort(ids.begin(), ids.end());

        // Reconstruir string de lambda formateado
        long long int_part = key / 1000000;
        long long frac_part = key % 1000000;
        string s_frac = to_string(frac_part);
        while (s_frac.size() < 6) s_frac = "0" + s_frac;
        // Eliminar ceros sobrantes pero dejando al menos 1 dígito
        while (s_frac.size() > 1 && s_frac.back() == '0') s_frac.pop_back();

        cout << int_part << "." << s_frac << ":";
        for (int id : ids) cout << " " << id;
        cout << "\n";
    }

    return 0;
}
#12 ParenTalk
Certamen 2025-01 #E [resuelto por el agente]
Problema: Dada una cadena de longitud n (<= 10^5) compuesta exclusivamente por '(' y ')'. Detectar la longitud de la subcadena contigua más larga que sea una secuencia de paréntesis correctamente balanceada.
Técnica: Simulación con Pila (Longest Valid Parentheses). Apilar inicialmente el índice base -1. Al leer '(', apilar su índice. Al leer ')', desapilar; si la pila queda vacía, apilar el índice actual como nuevo centinela; de lo contrario, `max_len = max(max_len, i - pila.top())`.
Tiempo: O(n) Espacio: O(n) 
#include <iostream>
#include <string>
#include <stack>
#include <algorithm>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    string s;
    if (!(cin >> n >> s)) return 0;

    stack<int> st;
    st.push(-1); // Centinela base
    int max_len = 0;

    for (int i = 0; i < n; i++) {
        if (s[i] == '(') {
            st.push(i);
        } else {
            st.pop();
            if (st.empty()) {
                st.push(i); // Nuevo inicio de subcadena válida
            } else {

#include <vector>
#include <algorithm>
using namespace std;

struct Op { int type, idx, color; };

void solve() {
    int n, m, c;
    if (!(cin >> n >> m >> c)) return;

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) cin >> g[i][j];

    vector<bool> row_cleared(n, false), col_cleared(m, false);
    vector<Op> ops;
    int remaining = n * m;

    while (remaining > 0) {
        bool found = false;
        // Buscar fila monocromática

                break;

                }

                    if (g[i][j] != 0) { g[i][j] = 0; remaining--; }
                }
                found = true;
                break;
            }
        }
        if (!found) break; // Bloqueado
    }

        cout << "YES\n" << ops.size() << "\n";
        reverse(ops.begin(), ops.end()); // Invertir orden del proceso hacia adelante
        for (const auto& op : ops) {

    }

#include <vector>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, m;
    if (!(cin >> n >> m)) return 0;
    vector<string> var(n), ref(m);
    for (int i = 0; i < n; i++) cin >> var[i];
    for (int i = 0; i < m; i++) cin >> ref[i];

    auto match_cost = [](const string& a, const string& b) {
        if (a == b) return 0;
        if (a == "UNK" || b == "UNK") return 1;
        return 5;
    };

    // dp[i][j] = costo de alinear var[0..i-1] con ref[0..j-1]
    vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
    for (int i = 0; i <= n; i++) dp[i][0] = 2 * i;
    for (int j = 0; j <= m; j++) dp[0][j] = 2 * j;

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            int c_del_var = dp[i - 1][j] + 2;
            int c_del_ref = dp[i][j - 1] + 2;
            int c_match   = dp[i - 1][j - 1] + match_cost(var[i - 1], ref[j - 1]);
            dp[i][j] = min({c_del_var, c_del_ref, c_match});
        }
    }

    cout << dp[n][m] << "\n";
    return 0;
}

    if (s[idx] != '_') {
        if (is_vowel(s[idx])) total += backtrack(idx + 1, vowels + 1, 0, has_l);

        // Opción 3: Colocar otra consonante (20 opciones)
        total += 20 * backtrack(idx + 1, 0, consonants + 1, has_l);
    }
    return total;
}

bool eval_wff() {

long long merge_and_count(vector<int>& arr, int l, int m, int r) {
    vector<int> left(arr.begin() + l, arr.begin() + m + 1);
    vector<int> right(arr.begin() + m + 1, arr.begin() + r + 1);
    int i = 0, j = 0, k = l;
    long long inv_count = 0;

    while (i < (int)left.size() && j < (int)right.size()) {
        if (left[i] <= right[j]) {
            arr[k++] = left[i++];
        } else {
            arr[k++] = right[j++];
            inv_count += (left.size() - i); // Todos los elementos restantes en left son inversiones
        }
    }
    while (i < (int)left.size()) arr[k++] = left[i++];
    while (j < (int)right.size()) arr[k++] = right[j++];
    return inv_count;
}

        count += mergesort_count(arr, m + 1, r);
        count += merge_and_count(arr, l, m, r);
    }
    return count;
}

#include <vector>
#include <queue>
using namespace std;

Tiempo: O(n^3 log x) Espacio: O(n^2) 
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

using ull = unsigned long long;

vector<vector<ull>> multiply(const vector<vector<ull>>& A, const vector<vector<ull>>& B, int n) {
    vector<vector<ull>> C(n, vector<ull>(n, 0));
    for (int i = 0; i < n; i++) {
        for (int k = 0; k < n; k++) {
            if (!A[i][k]) continue;
            for (int j = 0; j < n; j++) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }
    return C;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n, x;
    if (!(cin >> n >> x)) return 0;

    // adj[u][v] = cantidad de pociones para pasar de u a v
    vector<vector<ull>> adj(n, vector<ull>(n, 0));
    for (int i = 0; i < n; i++) {
        int s; cin >> s;
        for (int k = 0; k < s; k++) {
            int src; cin >> src;
            adj[src][i]++; // De src se transforma en i
        }
    }

    // Exponenciación binaria de matriz: res = adj^x
    vector<vector<ull>> res(n, vector<ull>(n, 0));
    for (int i = 0; i < n; i++) res[i][i] = 1;
    vector<vector<ull>> base = adj;
    int p = x;
    while (p > 0) {
        if (p & 1) res = multiply(res, base, n);
        base = multiply(base, base, n);
        p >>= 1;
    }

    // Buscar el par (a, b) con el máximo potencial
    ull max_pot = 0;
    int best_a = 0, best_b = 0;
    for (int a = 0; a < n; a++) {
        for (int b = 0; b < n; b++) {
            if (res[a][b] > max_pot) {
                max_pot = res[a][b];
                best_a = a;
                best_b = b;
            }
        }
    }

    cout << best_a << " " << best_b << " " << max_pot << "\n";
    return 0;
}
