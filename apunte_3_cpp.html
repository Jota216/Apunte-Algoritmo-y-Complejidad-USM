¿Por qué Fast I/O? ios::sync_with_stdio(false) desvincula los buffers de C++ (cin/cout) de los de C (scanf/printf). cin.tie(nullptr) evita que cin vacíe cout antes de cada lectura. En $10^6$ enteros, reduce tiempo de ~1.4s a ~0.08s.
¡REGLA DE ORO! Al usar Fast I/O, NUNCA mezcles cin/cout con scanf/printf/getchar. Usar ' ' en vez de endl (pues endl fuerza flush innecesario).

1.4 std::deque<T> (Doble Cola) O(1) Extremos

Buffer de bloques continuos. Inserción y borrado en ambos extremos en $O(1)$, con acceso aleatorio dq[i] en $O(1)$.

dq.push_front(20); // Inserta al frente O(1)

Patrón 1: 0-1 BFS con Deque (Calle Mágica C2_2024_02): Para grafos con aristas de peso 0 y 1. Si peso es 0 → push_front; si es 1 → push_back. Encuentra caminos mínimos en $O(V + E)$ sin Dijkstra. 
vector<int> dist(n, INF);
deque<int> dq;
dist[src] = 0;
dq.push_back(src);
while (!dq.empty()) {
    int u = dq.front(); dq.pop_front();
    for (auto& [v, w] : adj[u]) {
        if (dist[u] + w < dist[v]) {
            dist[v] = dist[u] + w;
            if (w == 0) dq.push_front(v);
            else        dq.push_back(v);
        }
    }
}

    while (!dq.empty() && a[dq.back()] >= a[i]) dq.pop_back(); // Mantener monotonía
    dq.push_back(i);
    if (i >= k - 1) min_window.push_back(a[dq.front()]); // Mínimo actual
}

1.6 std::priority_queue<T> (Heap Binario) O(log N)

Cola de prioridad basada en montículo binario. Acceso al elemento óptimo en $O(1)$, inserción y extracción en $O(log N)$.

max_pq.pop();         // Elimina 50

priority_queue<int, vector<int>, greater<int>> min_pq;

min_pq.pop();         // Elimina 10

// 4. Con Structs y operador < sobrecargado:
struct Tarea {
    int id, prioridad, duracion;
    // OJO: Para MIN-HEAP se invierte la lógica (< retorna mayor):

struct DSU {
    vector<int> parent, sz;
    int num_sets;

    DSU(int n) : parent(n), sz(n, 1), num_sets(n) {
        iota(parent.begin(), parent.end(), 0);
    }

    int find(int i) {
        if (parent[i] == i) return i;
        return parent[i] = find(parent[i]); // Compresión de camino
    }

    bool unite(int i, int j) {
        int root_i = find(i);
        int root_j = find(j);
        if (root_i == root_j) return false; // Ya estaban conectados

        // Unión por tamaño
        if (sz[root_i] < sz[root_j]) swap(root_i, root_j);
        parent[root_j] = root_i;
        sz[root_i] += sz[root_j];
        num_sets--;
        return true;
    }

    bool same(int i, int j) { return find(i) == find(j); }
    int size(int i) { return sz[find(i)]; }
};

3.8 Segment Tree Iterativo (Range Minimum / Sum Query) O(log N) Rango

Estructura iterativa en base 0, no recursiva, ultra-rápida (tamaño $2N$):

struct SegTree {
    int n;
    vector<long long> t;
    SegTree(int n) : n(n), t(2 * n, INF64) {} // Para sumas iniciar en 0

    void build(const vector<long long>& a) {
        for (int i = 0; i < n; ++i) t[n + i] = a[i];
        for (int i = n - 1; i > 0; --i) t[i] = min(t[i << 1], t[i << 1 | 1]);
    }

    void update(int p, long long val) { // Asignación puntual en índice p
        for (t[p += n] = val; p > 1; p >>= 1)
            t[p >> 1] = min(t[p], t[p ^ 1]);
    }

    long long query(int l, int r) { // Consulta mínimo en rango [l, r]
        long long res = INF64;
        for (l += n, r += n + 1; l < r; l >>= 1, r >>= 1) {
            if (l & 1) res = min(res, t[l++]);
            if (r & 1) res = min(res, t[--r]);
        }
        return res;
    }
};

                            dp[i][j - 1] + gap});           // Gap en X (izquierda)

    // Poda por cota: si el costo acumulado ya supera o iguala la mejor solución:
    if (current_cost >= best_cost) return;

    if (count == n) {
        // Retorno al origen (nodo 0):
        if (adj_matrix[u][0] < INF64) {
            best_cost = min(best_cost, current_cost + adj_matrix[u][0]);
        }
        return;
    }

    for (int v = 0; v < n; ++v) {
        if (!vis[v] && adj_matrix[u][v] < INF64) {
            vis[v] = true;
            tsp(v, count + 1, current_cost + adj_matrix[u][v]);
            vis[v] = false; // Backtrack
        }
    }
}

3.29 String Hashing Polinomial en O(1) Subcadenas

struct StringHash {
    const long long P = 313, MOD = 1e9 + 9;
    vector<long long> h, p_pow;

    StringHash(const string& s) {
        int n = s.size();
        h.assign(n + 1, 0); p_pow.assign(n + 1, 1);
        for (int i = 0; i < n; ++i) {
            p_pow[i + 1] = (p_pow[i] * P) % MOD;
            h[i + 1] = (h[i] * P + s[i]) % MOD;
        }
    }

    // Retorna el hash del rango s[l..r] inclusivo en O(1):
    long long get(int l, int r) {
        long long res = (h[r + 1] - h[l] * p_pow[r - l + 1]) % MOD;
        return (res + MOD) % MOD;
    }
};

4.2 Árbol de Decisión: ¿Qué Contenedor Elegir? Elección Rápida
  • "Necesito acceso aleatorio rápido por índice entero": → vector<T>.
  • "Necesito agregar/sacar eficientemente en ambos extremos": → deque<T> (0-1 BFS).
  • "Necesito el elemento máximo o mínimo continuamente": → priority_queue<T> (Dijkstra, Prim).
  • "Necesito mantener elementos ordenados sin duplicados y consultar cotas": → set<T> (lower_bound $O(log N)$).
  • "Necesito mantener elementos ordenados con duplicados": → multiset<T> (ojo con erase(find(x))).
  • "Necesito asociar claves arbitrarias (strings, coordenadas) a valores": → map (ordenado) o unordered_map (rápido $O(1)$).
  • "Necesito insertar en el medio en $O(1)$ mediante punteros/cursores": → list<T> (Simulación de texto, Sim.cpp).
  • "Necesito subconjuntos de hasta 30 elementos": → Bitmask entero (int o long long).
  • "Necesito operaciones de conjuntos de hasta $10^5$ booleanos": → bitset<N> (Mochila en $O(NW/64)$).
